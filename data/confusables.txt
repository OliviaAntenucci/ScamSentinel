# Curated subset of Unicode confusables (UTS #39) for lookalike-domain detection.
# Format: <code point> <ASCII replacement>  # <character name>
# Code points are written in hex so invisible/lookalike characters never hide in this file.
# NFKC already folds fullwidth and mathematical alphanumeric forms, and combining marks
# (ü, é, ...) are stripped separately, so they are not listed here.

# Cyrillic
U+0430 a  # CYRILLIC SMALL LETTER A
U+0432 b  # CYRILLIC SMALL LETTER VE (in some fonts)
U+0435 e  # CYRILLIC SMALL LETTER IE
U+043E o  # CYRILLIC SMALL LETTER O
U+0440 p  # CYRILLIC SMALL LETTER ER
U+0441 c  # CYRILLIC SMALL LETTER ES
U+0443 y  # CYRILLIC SMALL LETTER U
U+0445 x  # CYRILLIC SMALL LETTER HA
U+0456 i  # CYRILLIC SMALL LETTER BYELORUSSIAN-UKRAINIAN I
U+0458 j  # CYRILLIC SMALL LETTER JE
U+0455 s  # CYRILLIC SMALL LETTER DZE
U+04BB h  # CYRILLIC SMALL LETTER SHHA
U+0501 d  # CYRILLIC SMALL LETTER KOMI DE
U+051B q  # CYRILLIC SMALL LETTER QA
U+051D w  # CYRILLIC SMALL LETTER WE
U+04CF l  # CYRILLIC SMALL LETTER PALOCHKA
U+043A k  # CYRILLIC SMALL LETTER KA
U+0442 t  # CYRILLIC SMALL LETTER TE (in some fonts)
U+043C m  # CYRILLIC SMALL LETTER EM (in some fonts)
U+043D h  # CYRILLIC SMALL LETTER EN (in some fonts)
U+0433 r  # CYRILLIC SMALL LETTER GHE (in some fonts)
U+0491 r  # CYRILLIC SMALL LETTER GHE WITH UPTURN
